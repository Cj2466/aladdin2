numbers (this is the same fact recorded in project memory as "sampling frequency does not shorten
detection"). Using 252 rather than 365 for the crypto rows therefore changes nothing.

  import sys; sys.path.insert(0,'.')
  from app.services.research_lab.sourcing_power_check import sourcing_power_check, PROCEED
  def minsr(years, nl, fracs):
      lo,hi=0.01,12.0
      for _ in range(20):
          mid=(lo+hi)/2
          r=sourcing_power_check(claimed_sharpe_annualized=mid, periods_per_year=252,
                                 years_of_data=years, n_local=nl, offset_fractions=fracs)
          if r.verdict==PROCEED: hi=mid
          else: lo=mid
      return hi
